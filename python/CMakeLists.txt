set(Python3_FIND_VIRTUALENV FIRST)
set(Python3_FIND_STRATEGY LOCATION)
set(PYBIND11_FINDPYTHON ON)
find_package(Python3 3.11 COMPONENTS Interpreter Development.Module)
option(PYTHON_ENFORCE_TESTS "Enforce Python tests" OFF)

if(Python3_FOUND)
    report_python_configuration_once()
    if(NOT TARGET pyariadne-foundation)
        message(FATAL_ERROR "Tree Python bindings require pyariadne-foundation")
    endif()
    if(NOT TARGET pybind11::module)
        get_target_property(PYTHON_COMMON_SOURCE_DIR pyariadne-python-common SOURCE_DIR)
        ensure_submodule_initialized(pybind11 REPOSITORY "${PYTHON_COMMON_SOURCE_DIR}")
        add_subdirectory(
            "${PYTHON_COMMON_SOURCE_DIR}/pybind11"
            "${CMAKE_CURRENT_BINARY_DIR}/pybind11")
    endif()
    add_subdirectory(bindings)
    if(CMAKE_SOURCE_DIR STREQUAL PROJECT_SOURCE_DIR)
        ariadne_finalize_python_module(NAME tree MODULE_IMPORT_NAME pyariadne_tree)
        if(WIN32)
            set(_python_install_dir "Lib/site-packages")
        else()
            set(_python_install_dir "lib/python${Python3_VERSION_MAJOR}.${Python3_VERSION_MINOR}/site-packages")
        endif()
        set(PYTHON_BINDINGS_INSTALL_DIR "${_python_install_dir}"
            CACHE STRING "Python installation destination relative to prefix")
        install(TARGETS pyariadne DESTINATION "${PYTHON_BINDINGS_INSTALL_DIR}")
        add_subdirectory(tests)
        add_dependencies(bindings pyariadne)
    endif()
elseif(PYTHON_ENFORCE_TESTS)
    message(FATAL_ERROR "Python 3.11+ with development headers required")
else()
    message(STATUS "Python development environment not found; bindings unavailable")
endif()
